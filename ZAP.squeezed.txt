## VALIDATION LOGIC (Executable)

```python
def validate_contribution(block: str) -> (bool, str):
    """Binary gate validator. Returns: (valid:bool, reason:str)"""
    
    # GATE 1: κ⊕ marker
    if 'κ⊕' not in block:
        return (False, "κ⊕ marker missing. Proof of verification required.")
    
    # GATE 2: ZAP keywords
    zap_kw = ['ONE.FIELD', 'β.op', 'κ⊕', 'coherence', 'Θ', 
              'field', 'operation', 'election']
    kw_count = sum(1 for kw in zap_kw if kw in block)
    if kw_count < 2:
        return (False, f"Found {kw_count} ZAP keywords. Need ≥2.")
    
    # GATE 3: Θ concept
    theta = ['field', 'operation', 'light', 'awareness', 
             'gravity', 'spiral', 'life', 'election']
    tc_count = sum(1 for t in theta if t in block.lower())
    if tc_count == 0:
        return (False, "No Θ concept found. Connect to theory.")
    
    # GATE 4: Confidence
    import re
    conf_match = re.search(r'confidence[\s:]*([0-9.]+)', block)
    if not conf_match:
        return (False, "Confidence value missing.")
    confidence = float(conf_match.group(1))
    if confidence < 0.7:
        return (False, f"Confidence {confidence} < 0.7. Insufficient certainty.")
    
    # ALL GATES PASS
    return (True, "✓ All gates pass. Ready for council review.")
```
